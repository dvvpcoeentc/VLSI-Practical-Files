
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity KEYPADTB_vhd is
end KEYPADTB_vhd;

architecture behavior of KEYPADTB_vhd is

    component matrix_keypad
        port (
            clock    : in  STD_LOGIC;
            rst      : in  STD_LOGIC;
            row_line : in  STD_LOGIC_VECTOR(3 downto 0);
            col_line : out STD_LOGIC_VECTOR(3 downto 0);
            d        : out STD_LOGIC_VECTOR(6 downto 0)
        );
    end component;

    -- Inputs
    signal clock    : STD_LOGIC := '0';
    signal rst      : STD_LOGIC := '0';
    signal row_line : STD_LOGIC_VECTOR(3 downto 0) := (others => '0');

    -- Outputs
    signal col_line : STD_LOGIC_VECTOR(3 downto 0);
    signal d        : STD_LOGIC_VECTOR(6 downto 0);

    constant clock_period : time := 20 ns;

begin

    -- Instantiate the Unit Under Test
    UUT: matrix_keypad
        port map (
            clock    => clock,
            rst      => rst,
            row_line => row_line,
            col_line => col_line,
            d        => d
        );

    -- Clock generation
    clock_process : process
    begin
        clock <= '0';
        wait for clock_period / 2;

        clock <= '1';
        wait for clock_period / 2;
    end process;

    -- Test stimulus
    tb : process
    begin

        -- Wait for 100 ns
        wait for 100 ns;

        rst <= '1';
        row_line <= "0101";

        wait for 20 ns;

        rst <= '0';

        wait;

    end process;

end behavior;
